import React, { useState } from 'react';
import { StyleSheet, Text, View, TextInput, Button, FlatList, ScrollView } from 'react-native';

export default function App() {
  const [activeSection, setActiveSection] = useState(null);

  // চিকিৎসা পরামর্শ
  const [disease, setDisease] = useState('');
  const [advice, setAdvice] = useState('');

  const getAdvice = () => {
    const dis = disease.toLowerCase();
    if (dis.includes('সর্দি')) setAdvice('হালকা সর্দি হলে বিশ্রাম নিন, গরম পানি পান করুন।');
    else if (dis.includes('জ্বর')) setAdvice('জ্বর থাকলে বিশ্রাম এবং হালকা খাবার গ্রহণ করুন।');
    else setAdvice('দয়া করে বিস্তারিত রোগ নাম লিখুন।');
  };

  // ঔষধ সংরক্ষণ
  const [medName, setMedName] = useState('');
  const [medQty, setMedQty] = useState('');
  const [medPrice, setMedPrice] = useState('');
  const [medicines, setMedicines] = useState([]);

  const addMedicine = () => {
    if (!medName || !medQty || !medPrice) return;
    const now = new Date();
    const date = now.toLocaleDateString('bn-BD');
    const time = now.toLocaleTimeString('bn-BD');

    setMedicines(prev => {
      const existing = prev.find(m => m.name === medName);
      if (existing) {
        return prev.map(m =>
          m.name === medName
            ? { ...m, qty: parseInt(m.qty) + parseInt(mQty), price: parseFloat(mPrice), date, time }
            : m
        );
      } else {
        return [...prev, { name: medName, qty: parseInt(medQty), price: parseFloat(mPrice), date, time }];
      }
    });

    setMedName(''); setMedQty(''); setMedPrice('');
  };

  // আয়-ব্যয় হিসাব
  const [income, setIncome] = useState('');
  const [expense, setExpense] = useState('');
  const [totalIncome, setTotalIncome] = useState(0);
  const [totalExpense, setTotalExpense] = useState(0);
  const [transactions, setTransactions] = useState([]);

  const addIncome = () => {
    if (!income) return;
    const val = parseFloat(income);
    const now = new Date();
    const date = now.toLocaleDateString('bn-BD');
    const time = now.toLocaleTimeString('bn-BD');
    const balance = totalIncome + val - totalExpense;

    setTransactions(prev => [...prev, { date, time, desc: 'আয়', income: val, expense: 0, balance }]);
    setTotalIncome(prev => prev + val);
    setIncome('');
  };

  const addExpense = () => {
    if (!expense) return;
    const val = parseFloat(expense);
    const now = new Date();
    const date = now.toLocaleDateString('bn-BD');
    const time = now.toLocaleTimeString('bn-BD');
    const balance = totalIncome - (totalExpense + val);

    setTransactions(prev => [...prev, { date, time, desc: 'ব্যয়', income: 0, expense: val, balance }]);
    setTotalExpense(prev => prev + val);
    setExpense('');
  };

  return (
    <ScrollView style={styles.container}>
      <Text style={styles.title}>The Ramu Homoeo Hall</Text>

      <View style={styles.buttonContainer}>
        <Button title="চিকিৎসা পরামর্শ" onPress={() => setActiveSection('consult')} />
        <Button title="ঔষধ সংরক্ষণ" onPress={() => setActiveSection('medicine')} />
        <Button title="আয়-ব্যয় হিসাব" onPress={() => setActiveSection('balance')} />
      </View>

      {activeSection === 'consult' && (
        <View style={styles.section}>
          <Text style={styles.sectionTitle}>চিকিৎসা পরামর্শ</Text>
          <TextInput style={styles.input} placeholder="রোগের নাম লিখুন" value={disease} onChangeText={setDisease} />
          <Button title="পরামর্শ দেখাও" onPress={getAdvice} />
          <Text style={{marginTop:10}}>{advice}</Text>
        </View>
      )}

      {activeSection === 'medicine' && (
        <View style={styles.section}>
          <Text style={styles.sectionTitle}>ঔষধ সংরক্ষণ ও বিক্রয়</Text>
          <TextInput style={styles.input} placeholder="ঔষধের নাম" value={medName} onChangeText={setMedName} />
          <TextInput style={styles.input} placeholder="পরিমাণ" value={medQty} onChangeText={setMedQty} keyboardType="numeric"/>
          <TextInput style={styles.input} placeholder="মূল্য" value={medPrice} onChangeText={setMedPrice} keyboardType="numeric"/>
          <Button title="সংরক্ষণ / আপডেট" onPress={addMedicine} />
          <FlatList
            data={medicines}
            keyExtractor={(item,index) => index.toString()}
            renderItem={({item}) => (
              <Text>{item.name} - {item.qty} pcs - {item.price} টাকা - {item.date} {item.time}</Text>
            )}
            style={{marginTop:10}}
          />
        </View>
      )}

      {activeSection === 'balance' && (
        <View style={styles.section}>
          <Text style={styles.sectionTitle}>আয়-ব্যয় ও ব্যালান্স</Text>
          <TextInput style={styles.input} placeholder="আয় লিখুন" value={income} onChangeText={setIncome} keyboardType="numeric"/>
          <Button title="যোগ করো" onPress={addIncome} />
          <TextInput style={styles.input} placeholder="ব্যয় লিখুন" value={expense} onChangeText={setExpense} keyboardType="numeric"/>
          <Button title="যোগ করো" onPress={addExpense} />
          <FlatList
            data={transactions}
            keyExtractor={(item,index)=>index.toString()}
            renderItem={({item})=>(
              <Text>{item.date} {item.time} - {item.desc} - {item.income}/{item.expense} - ব্যালান্স: {item.balance}</Text>
            )}
            style={{marginTop:10}}
          />
          <Text style={styles.balance}>ব্যালান্স: {totalIncome - totalExpense} টাকা</Text>
        </View>
      )}
    </ScrollView>
  );
}

const styles = StyleSheet.create({
  container:{ flex:1, backgroundColor:'#f5f5f5', padding:10 },
  title:{ fontSize:24, fontWeight:'bold', textAlign:'center', color:'#1a237e', marginBottom:20 },
  buttonContainer:{ flexDirection:'row', justifyContent:'space-around', marginBottom:20 },
  section:{ backgroundColor:'white', padding:15, borderRadius:10, marginBottom:20, shadowColor:'#000', shadowOffset:{width:0,height:2}, shadowOpacity:0.3, shadowRadius:3, elevation:3 },
  sectionTitle:{ fontSize:18, fontWeight:'bold', marginBottom:10 },
  input:{ borderWidth:1, borderColor:'#ccc', borderRadius:5, padding:8, marginBottom:10 },
  balance:{ fontWeight:'bold', fontSize:16, color:'#1a237e', marginTop:10 }
});
